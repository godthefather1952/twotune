cmake_minimum_required(VERSION 3.22)
project(EvolutionVox VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(EVOLUTIONVOX_BUILD_PLUGIN "Build the VST3 plugin using JUCE" ON)
option(EVOLUTIONVOX_BUILD_TESTS "Build standalone DSP unit tests" ON)
add_library(EvolutionVoxDSP INTERFACE)
target_include_directories(EvolutionVoxDSP INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/Source)
if(EVOLUTIONVOX_BUILD_TESTS)
  enable_testing()
  add_executable(evolutionvox_dsp_tests Tests/test_dsp.cpp)
  target_link_libraries(evolutionvox_dsp_tests PRIVATE EvolutionVoxDSP)
  add_test(NAME evolutionvox_dsp_tests COMMAND evolutionvox_dsp_tests)
  add_executable(evolutionvox_dial_mapping_tests Tests/test_dial_mapping.cpp)
  target_link_libraries(evolutionvox_dial_mapping_tests PRIVATE EvolutionVoxDSP)
  add_test(NAME evolutionvox_dial_mapping_tests COMMAND evolutionvox_dial_mapping_tests)
  add_executable(evolutionvox_zeus_rig_tests Tests/test_zeus_rig.cpp)
  target_link_libraries(evolutionvox_zeus_rig_tests PRIVATE EvolutionVoxDSP)
  add_test(NAME evolutionvox_zeus_rig_tests COMMAND evolutionvox_zeus_rig_tests)
  add_executable(evolutionvox_demo Tests/render_demo.cpp)
  target_link_libraries(evolutionvox_demo PRIVATE EvolutionVoxDSP)
  # Manual DSP benchmark (not a DAW CPU measurement); not registered with ctest.
  add_executable(evolutionvox_perf_bench Tests/perf_bench.cpp)
  target_link_libraries(evolutionvox_perf_bench PRIVATE EvolutionVoxDSP)
endif()
if(EVOLUTIONVOX_BUILD_PLUGIN)
  # With JUCE installed locally: cmake -S . -B build -DFETCH_JUCE=OFF -DJUCE_DIR=/path/to/JUCE
  # Otherwise CMake downloads the pinned JUCE source during configuration.
  option(FETCH_JUCE "Fetch JUCE when not already installed" ON)
  if(DEFINED JUCE_DIR AND EXISTS "${JUCE_DIR}/CMakeLists.txt")
    add_subdirectory("${JUCE_DIR}" juce)
  elseif(FETCH_JUCE)
    include(FetchContent)
    FetchContent_Declare(juce GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.17-security GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(juce)
  else()
    message(FATAL_ERROR "JUCE not found. Supply -DJUCE_DIR=/path/to/JUCE or -DFETCH_JUCE=ON")
  endif()

  # Optimized Zeus layers generated by tools/build_zeus_assets.py (sources in Assets/Zeus/source).
  file(GLOB EVOLUTIONVOX_ZEUS_PNGS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/Assets/Zeus/ui/*.png)
  juce_add_binary_data(EvolutionVoxAssets HEADER_NAME BinaryData.h NAMESPACE BinaryData
      SOURCES ${EVOLUTIONVOX_ZEUS_PNGS})

  # Plugin sources shared by the VST3 and the JUCE-level plugin tests.
  set(EVOLUTIONVOX_PLUGIN_SOURCES
      ${CMAKE_CURRENT_SOURCE_DIR}/Source/Plugin/PluginProcessor.cpp
      ${CMAKE_CURRENT_SOURCE_DIR}/Source/Plugin/PluginEditor.cpp
      ${CMAKE_CURRENT_SOURCE_DIR}/Source/Plugin/LightningRetuneDial.cpp
      ${CMAKE_CURRENT_SOURCE_DIR}/Source/Plugin/SettingsPanel.cpp)
  set(EVOLUTIONVOX_JUCE_LIBS
      EvolutionVoxDSP
      EvolutionVoxAssets
      juce::juce_audio_utils
      juce::juce_audio_processors
      juce::juce_dsp
      juce::juce_recommended_config_flags
      juce::juce_recommended_lto_flags
      juce::juce_recommended_warning_flags)

  juce_add_plugin(EvolutionVox
      COMPANY_NAME "EvolutionVox"
      BUNDLE_ID "audio.evolutionvox.plugin"
      PLUGIN_MANUFACTURER_CODE Evol
      PLUGIN_CODE Evo1
      # VST3 for DAWs + Standalone app (same processor and editor) for testing without a DAW.
      FORMATS VST3 Standalone
      VST3_CAN_REPLACE_VST2 FALSE
      # macOS requires a usage string before the standalone app may open the microphone.
      MICROPHONE_PERMISSION_ENABLED TRUE
      MICROPHONE_PERMISSION_TEXT "EvolutionVox tunes the vocal coming from your audio input."
      PRODUCT_NAME "EvolutionVox"
      IS_SYNTH FALSE
      NEEDS_MIDI_INPUT FALSE
      NEEDS_MIDI_OUTPUT FALSE
      IS_MIDI_EFFECT FALSE
      COPY_PLUGIN_AFTER_BUILD FALSE)
  target_sources(EvolutionVox PRIVATE ${EVOLUTIONVOX_PLUGIN_SOURCES})
  target_include_directories(EvolutionVox PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/Source/Plugin)
  target_link_libraries(EvolutionVox PRIVATE ${EVOLUTIONVOX_JUCE_LIBS})
  target_compile_definitions(EvolutionVox PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
  juce_generate_juce_header(EvolutionVox)

  if(EVOLUTIONVOX_BUILD_TESTS)
    # Processor/editor tests: parameter IDs, state recall + migration, power/bypass latency,
    # key/scale reaching the DSP, offline CPU timing, and editor screenshots (--screenshots DIR).
    juce_add_console_app(EvolutionVoxPluginTests PRODUCT_NAME "EvolutionVoxPluginTests")
    target_sources(EvolutionVoxPluginTests PRIVATE Tests/plugin_tests.cpp ${EVOLUTIONVOX_PLUGIN_SOURCES})
    target_include_directories(EvolutionVoxPluginTests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/Source/Plugin)
    target_link_libraries(EvolutionVoxPluginTests PRIVATE ${EVOLUTIONVOX_JUCE_LIBS})
    target_compile_definitions(EvolutionVoxPluginTests PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0
        JucePlugin_Name="EvolutionVox")
    juce_generate_juce_header(EvolutionVoxPluginTests)
    add_test(NAME evolutionvox_plugin_tests COMMAND EvolutionVoxPluginTests)
  endif()
endif()
